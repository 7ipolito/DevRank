# World ID App ID (get from https://developer.worldcoin.org)

# Your Competition Contract Address on Worldchain Mainnet
# Replace with your deployed contract address
NEXT_PUBLIC_COMPETITION_CONTRACT_ADDRESS=0xYOUR_CONTRACT_ADDRESS_HERE

# OPTIONAL VARIABLES (Recommended for Production)
# ============================================

# Worldchain RPC Endpoint
# Get free API key from: https://www.alchemy.com/
# Free tier: 100k requests/day
# Format: https://worldchain-mainnet.g.alchemy.com/v2/YOUR_API_KEY
# If not set, uses public endpoint (may be rate limited)
NEXT_PUBLIC_WORLDCHAIN_RPC_URL=https://worldchain-mainnet.g.alchemy.com/v2/YOUR_ALCHEMY_API_KEY
